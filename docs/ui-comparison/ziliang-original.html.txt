      const escapeHtml = (value) => String(value).replace(/[&<>"']/g, (char) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[char]));
      function renderDetail(node) {
        if (!node) { detail.innerHTML = '<p class="prototype-inline-empty">选择左侧节点后，这里显示负责人、证据、预期与下一步。</p>'; return; }
        detail.innerHTML = `<div class="wbs-detail-title"><span class="wbs-detail-status ${node.status}">${escapeHtml(node.statusLabel)}</span><h4>${escapeHtml(node.title)}</h4><p>${escapeHtml(node.owner)}</p></div><dl class="wbs-detail-list"><div><dt>发布时间</dt><dd>${escapeHtml(node.publishedAt)}</dd></div><div><dt>发布人</dt><dd>${escapeHtml(node.publisher)}</dd></div><div><dt>发布给</dt><dd>${escapeHtml(node.recipient)}</dd></div><div><dt>完成者</dt><dd>${escapeHtml(node.completedBy)}</dd></div><div><dt>完成时间</dt><dd>${escapeHtml(node.completedAt)}</dd></div><div class="wbs-effect-row"><dt>执行效果</dt><dd>${escapeHtml(node.effect)}</dd></div><div><dt>实际结果</dt><dd>${escapeHtml(node.evidence)}</dd></div><div><dt>下一步</dt><dd>${escapeHtml(node.next)}</dd></div><div><dt>验收边界</dt><dd>仅记录本节点已验证范围，不从局部通过推断其他宿主或完整任务通过。</dd></div></dl><p class="wbs-detail-note">本地展示数据，不会调用 Agent 工具、发送消息或修改星云驿任务。</p>`;
      }
      function render(filter = 'all') {
        tree.innerHTML = workstreams.map((stream) => {
          const children = stream.children.filter((node) => filter === 'all' || (filter === 'attention' ? node.status !== 'done' : node.status === 'done'));
          if (!children.length) return '';
          return `<article class="wbs-workstream ${stream.open ? 'open' : ''}" data-stream="${stream.id}"><button type="button" class="wbs-workstream-head" aria-expanded="${stream.open}"><span class="wbs-toggle" aria-hidden="true">›</span><span class="wbs-workstream-copy"><strong>${escapeHtml(stream.title)}</strong><small>${escapeHtml(stream.summary)}</small></span><span class="wbs-workstream-meta"><span class="wbs-state ${stream.state}">${escapeHtml(stream.stateLabel)}</span><span>${children.length} 项</span></span></button><div class="wbs-children">${children.map((node) => `<button type="button" class="wbs-node ${node.status}" data-node="${node.id}"><span class="wbs-node-dot" aria-hidden="true"></span><span class="wbs-node-copy"><strong>${escapeHtml(node.title)}</strong><small>${escapeHtml(node.owner)} · 发布 ${escapeHtml(node.publishedAt)}</small><small class="wbs-node-route">${escapeHtml(node.publisher)} → ${escapeHtml(node.recipient)} · 完成：${escapeHtml(node.completedBy)}${node.completedAt === '—' ? '' : ` · ${escapeHtml(node.completedAt)}`}</small><small class="wbs-node-effect">效果：${escapeHtml(node.effect)}</small></span><span class="wbs-node-status ${node.status}">${escapeHtml(node.statusLabel)}</span></button>`).join('')}</div></article>`;
        }).join('');
        tree.querySelectorAll('.wbs-workstream-head').forEach((button) => button.addEventListener('click', () => { const parent = button.closest('.wbs-workstream'); parent.classList.toggle('open'); button.setAttribute('aria-expanded', parent.classList.contains('open')); }));
        tree.querySelectorAll('.wbs-node').forEach((button) => button.addEventListener('click', () => { tree.querySelectorAll('.wbs-node').forEach((node) => node.classList.remove('selected')); button.classList.add('selected'); const node = workstreams.flatMap((stream) => stream.children).find((item) => item.id === button.dataset.node); renderDetail(node); }));
      }
      document.querySelectorAll('.wbs-filter button').forEach((button) => button.addEventListener('click', () => { document.querySelectorAll('.wbs-filter button').forEach((item) => item.classList.remove('active')); button.classList.add('active'); render(button.dataset.filter); }));
      document.querySelectorAll('[data-wbs-target]').forEach((button) => button.addEventListener('click', () => { if (button.dataset.wbsFilter) document.querySelector(`.wbs-filter button[data-filter="${button.dataset.wbsFilter}"]`).click(); document.getElementById(button.dataset.wbsTarget).scrollIntoView({ behavior: 'smooth', block: 'start' }); }));
      render();
      renderDetail(null);
    </script>
  </body>
